   1. python $(python -c "import sys; print(sys.prefix)")/bin/pip-date.py
   2. python /Users/user/.pyenv/versions/smarty-notebook/bin/pip-date.py
   3. alias pip-date='python $(python -c "import sys; print(sys.prefix)")/bin/pip-date.py'

```
# Make the file executable
chmod +x $(python -c "import sys; print(sys.prefix)")/bin/pip-date.py

# Create a symlink without the .py extension
ln -s $(python -c "import sys; print(sys.prefix)")/bin/pip-date.py $(python -c "import sys; print(sys.prefix)")/bin/pip-date

# Rehash pyenv
pyenv rehash
```
